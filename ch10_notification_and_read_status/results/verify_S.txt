SELECT count(*) AS ab_personal_diff FROM (

\echo '=== (3) 告知の未読件数が案A と案B で一致すること ==='
=== (3) 告知の未読件数が案A と案B で一致すること ===
-- 案A は行を展開しているので read_at IS NULL を数える。
-- 案B は「告知はあるが読んだ記録が無い」を数える。
-- 🔴 この 2 つが一致することが、案B が案A と同じ答えを返す証拠である。
--    案C は畳んでいるので一致しない（それが案C の代償で、(4) で数える）。
WITH a AS (
  SELECT user_id, count(*) AS n FROM ch10_a.notifications
   WHERE kind = 'announcement' AND read_at IS NULL GROUP BY user_id
), b AS (
  SELECT u.id AS user_id, count(*) AS n
    FROM ch10_r.src_user u
   CROSS JOIN ch10_b.broadcasts bc
   WHERE NOT EXISTS (SELECT 1 FROM ch10_b.broadcast_reads r
                      WHERE r.broadcast_id = bc.id AND r.user_id = u.id)
   GROUP BY u.id
)
SELECT count(*) AS ab_broadcast_unread_diff FROM (
  (SELECT * FROM a EXCEPT ALL SELECT * FROM b)
  UNION ALL
  (SELECT * FROM b EXCEPT ALL SELECT * FROM a)
) t;

\echo '=== (4) 案C が畳んだことで失う既読（代償の大きさ） ==='
=== (4) 案C が畳んだことで失う既読（代償の大きさ） ===
-- 🔴 これは 0 にならないのが正しい。案C の代償を数値で示すための測定であって、
--    合否の検査ではない。だから先頭列は 0 を返す形にし、
--    実際の件数は 2 列目以降に出す。
SELECT 0 AS always_zero,
       count(*) AS lost_reads,
       round(100.0 * count(*) /
             nullif((SELECT count(*) FROM ch10_r.src_delivery d2
                      JOIN ch10_c.broadcasts b2 ON b2.id = d2.notice_id
                     WHERE d2.read_at IS NOT NULL), 0), 1) AS lost_pct
  FROM ch10_r.src_delivery d
  JOIN ch10_c.broadcasts b ON b.id = d.notice_id
  JOIN ch10_c.read_cursors c ON c.user_id = d.user_id
 WHERE d.read_at IS NOT NULL
   AND b.id > c.broadcast_read_upto;

=== (5) カーソルは既読を「未読側」にしか誤らないこと ===
-- 🔴 案C が安全側に倒れていることの確認。

SELECT count(*) AS swallowed_unread
  FROM ch10_c.broadcasts b

                    WHERE d.notice_id = b.id AND d.user_id = c.user_id
                      AND d.read_at IS NOT NULL);
