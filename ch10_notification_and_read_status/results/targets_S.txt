-- 🔴 num_nonnulls() で「ちょうど 1 つだけ非 NULL」を 1 行で書ける。
-- 🔴 そのうえで「種類」を生成列で導出できる。導出するので食い違いが起きない。
CREATE TABLE ch10_b.tgt_fk_cols (
  id          bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  order_id    bigint REFERENCES ch10_b.tgt_order(id),
  comment_id  bigint REFERENCES ch10_b.tgt_comment(id),
  target_kind text GENERATED ALWAYS AS (
    CASE WHEN order_id   IS NOT NULL THEN 'order'
         WHEN comment_id IS NOT NULL THEN 'comment' END) STORED,
  CONSTRAINT exactly_one_target CHECK (num_nonnulls(order_id, comment_id) = 1)
);
