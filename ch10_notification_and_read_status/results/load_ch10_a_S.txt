-- 🔴 全員向けの告知は、利用者の数だけ行を作る。これが案A の代償である。
--    元データの src_delivery は「読んだ人」しか持たないので、
--    読んでいない人の行はここで作る（左外部結合）。
INSERT INTO ch10_a.notifications (user_id, kind, body, read_at, created_at)
SELECT u.id, n.kind, n.body, d.read_at, n.created_at
  FROM ch10_r.src_notice n
 CROSS JOIN ch10_r.src_user u
  LEFT JOIN ch10_r.src_delivery d
         ON d.notice_id = n.id AND d.user_id = u.id
 WHERE n.is_broadcast
 ORDER BY n.id, u.id;
